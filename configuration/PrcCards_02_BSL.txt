// Функция "ПолучитьСостояние"
// Возвращает текущие значения реквизитов Карточек соответствующие условиям отбора, 
// и временные таблицы с этими данными в МенеджерВременныхТаблиц.
// Возвращает массив структур с полями:
//   Карточка, ВидРеквизита, УИД_Элемента, ЗначениеРеквизита, Хранение.
//
// Параметры:
//  "Отбор" - Соответствие. Содержит условия отбора.
//  "ФильтрВидовРеквизитов" - Массив. Содержит ссылки на ПланВидовХарактеристик._ВидыРеквизитов.
//     Если задан, в результат попадут только значения указанных реквизитов.
//  "МенеджерВременныхТаблиц" - МенеджерВременныхТаблиц.
//  "Рекурсия" - Число. Разрешённая глубина рекурсии для получения вложенных реквизитов (по умолчанию 3).
Функция ПолучитьСостояние(ЗНАЧ Отбор = Неопределено, ЗНАЧ ФильтрВидовРеквизитов = Неопределено, МенеджерВременныхТаблиц = Неопределено, ЗНАЧ Рекурсия = 3) Экспорт
	// 1. Подготовка Менеджера Временных Таблиц
	Если Не ТипЗнч(МенеджерВременныхТаблиц) = Тип("МенеджерВременныхТаблиц") Тогда
		МенеджерВременныхТаблиц = Новый МенеджерВременныхТаблиц;
	КонецЕсли;

	// 2. Инициализация переменных
	МассивКарточек = Новый Массив;
	ОтборыПоРеквизитам = Новый Соответствие;
	ВидКарточки = Неопределено;

	// 3.1 Обработка параметра "Отбор"
	Если Отбор <> Неопределено Тогда
		Для Каждого КлючЗнач Из Отбор Цикл
			Если КлючЗнач.Ключ = "Карточка" 
				И ТипЗнч(КлючЗнач.Значение) = Тип("СправочникСсылка._Карточки") Тогда
				Если МассивКарточек.Найти(КлючЗнач.Значение) = Неопределено Тогда
					МассивКарточек.Добавить(КлючЗнач.Значение);
				КонецЕсли;
			ИначеЕсли КлючЗнач.Ключ = "Карточки" 
				И ТипЗнч(КлючЗнач.Значение) = Тип("Массив") Тогда
				Для Каждого Карточка Из КлючЗнач.Значение Цикл
					Если МассивКарточек.Найти(Карточка) = Неопределено Тогда
						МассивКарточек.Добавить(Карточка);
					КонецЕсли;
				КонецЦикла;
			ИначеЕсли КлючЗнач.Ключ = "ВидКарточки" Тогда 
				ВидКарточки = КлючЗнач.Значение;
			ИначеЕсли ТипЗнч(КлючЗнач.Ключ) = Тип("ПланВидовХарактеристикСсылка._ВидыРеквизитов") Тогда
				ОтборыПоРеквизитам.Вставить(КлючЗнач.Ключ, КлючЗнач.Значение);
			КонецЕсли;
		КонецЦикла;
	КонецЕсли;

	// 4. Определение ВидКарточки
	// если вид карточки не указан, но установлен отбор по картачкам, буду использовать вид карточки из первой карточки.
	// Вид карточки должен быть указан явно, т.к. на основании вида карточки определяются реквизиты которые нужно извлекать
	Если ВидКарточки = Неопределено И МассивКарточек.Количество() > 0 Тогда
		ВидКарточки = МассивКарточек[0].ВидКарточки;
	КонецЕсли;

	Если ВидКарточки = Неопределено Тогда
		ВызватьИсключение "Не задан и не может быть определен параметр ВидКарточки в отборе или по карточкам.";

	ТекстЗапросаНаУдаление = "";

	// 5.01 ВТ_Реквизиты (шаблон вида карточки)
	// Левое соединение с этой таблицей гарантирует, что в результат попадут ТОЛЬКО реквизиты, 
	// описанные в ТЧ "Реквизиты" текущего ВидаКарточки.
	// 5.01 
	// ВТ_Реквизиты - список реквизитов значения, которых нужно вернуть в результате
	// ВТ_Реквизиты_Вычисляемые - список реквизитов значения, которых нужно вычислить. в этой таблице присутствуют реквизиты, которые будут использоваться в итерациях  
	ТекстЗапросаВТ_Реквизиты = "

		|ИЗ

		|	Ссылка = &ВидКарточки
		|;
		|
		|////////////////////////////////////////////////////////////////////////////////
		|ВЫБРАТЬ
		|	ВидКарточки КАК ВидКарточки,
		|	ВидРеквизита КАК ВидРеквизита,
		|	Хранение КАК Хранение
		|ПОМЕСТИТЬ ВТ_Реквизиты
		|ИЗ
		|	ВТ_Реквизиты_Все КАК Реквизиты
		|ГДЕ
		|	ВидРеквизита В (&МассивВидовРеквизитов)
		|;
		|
		|////////////////////////////////////////////////////////////////////////////////
		|ВЫБРАТЬ
		|	ВидКарточки КАК ВидКарточки,
		|	ВидРеквизита КАК ВидРеквизита,
		|	Хранение КАК Хранение
		// продумать логику, походу эта ВТ должна называться ВТ_Реквизиты_Для_Вычисления
		|ПОМЕСТИТЬ ВТ_Реквизиты_Вычисляемые
		|ИЗ
		|	ВТ_Реквизиты_Все КАК Реквизиты
		|ГДЕ
		|	(ВидРеквизита В (&МассивВидовРеквизитов)
		|	ИЛИ ВидРеквизита В (ВЫБРАТЬ ВТ_Реквизиты.Хранение ИЗ ВТ_Реквизиты))
		|;				
		|";

	Если ФильтрВидовРеквизитов <> Неопределено 
		И ТипЗнч(ФильтрВидовРеквизитов) = Тип("Массив") 
		И ФильтрВидовРеквизитов.Количество() > 0 
	Тогда
		ПараметрыЗапроса.Вставить("МассивВидовРеквизитов", ФильтрВидовРеквизитов);
	Иначе	
		ТекстЗапросаВТ_Реквизиты = СтрЗаменить(ТекстЗапросаВТ_Реквизиты,"ВидРеквизита В (&МассивВидовРеквизитов)","ИСТИНА");
	КонецЕсли;
	ТекстЗапросаНаУдаление = ТекстЗапросаНаУдаление + "
		|УНИЧТОЖИТЬ ВТ_Реквизиты_Все;
		|УНИЧТОЖИТЬ ВТ_Реквизиты_Вычисляемые;";

	ТекстЗапросаОтобранныхКарточек = "";
	ВТ_ТаблицаОтобраныхКарточек = "ВТ_Пустая";

	// 5.11 Фильтр по явно заданным карточкам
	Если МассивКарточек.Количество() > 0 Тогда
		Условие_МассивКарточек = " И Карточка В (&МассивКарточек)";
		ПараметрыЗапроса.Вставить("МассивКарточек", МассивКарточек);
		ТекстЗапросаОтобранныхКарточек = ТекстЗапросаОтобранныхКарточек + "
		|ВЫБРАТЬ
		|	_Карточки.Ссылка КАК Карточка
		|ПОМЕСТИТЬ ВТ_Карточки
		|ИЗ
		|	Справочник._Карточки КАК _Карточки
		|ГДЕ
		|	_Карточки.Ссылка В (&МассивКарточек)
		|;";
		ВТ_ТаблицаОтобраныхКарточек = "ВТ_Карточки";
		
		ТекстЗапросаНаУдаление = ТекстЗапросаНаУдаление + "
		|УНИЧТОЖИТЬ ВТ_Карточки;";
	Иначе

	КонецЕсли;

	// 5.12 Отбор по значениям реквизитов (логика И через ВНУТРЕННЕЕ СОЕДИНЕНИЕ)

			|	) КАК Регистр";
			
			Если Сч > 1 Тогда
				ТекстЗапросаОтобранныхКарточек = ТекстЗапросаОтобранныхКарточек + "
