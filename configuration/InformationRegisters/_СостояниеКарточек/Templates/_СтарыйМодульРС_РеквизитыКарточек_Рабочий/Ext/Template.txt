// Модуль менеджера регистра сведений: _РеквизитыКарточек

//Функция "ПолучитьСостояние"
//
// Возвращает текущие значения реквизитов Карточек соответствующие условиям отбора, и временные таблицы с этими данными.
// Возвращает массив структур с полями: 
///   Карточка (Справочник._Карточки.Ссылка), 
//    ВидРеквизита (Справочник._ВидыРеквизитов.Ссылка), 
//    УИД_Элемента (Уникальный индефикатор), используется для различения списочных реквизитов 
//    ЗначениеРеквизита (произвольный тип).
// в менеджере МенеджерВременныхТаблиц возвращаются таблицы ВТ_Реквизиты, ВТ_ЗначенияРеквизитов
// 
// Праметры:
//  "ВидКарточки" -Справочник._ВидыКарточек.Ссылка.  содержит Вид карточки, по которому выбираются реквизиты. 
//     Если не задан, тогда  в праметре "Отбор" должно присутствовать свойство "Карточка", что бы была возможность определить ВидКарточки на основании Карточки.
//  "Отбор" содержит условия отбора карточек. Может содержать как массив карточек так и значения реквизитов по которым нужно отбирать карточки.
//     Соответствие. 
//     где:
//         Ключ - "Карточка" (ссылка на _Карточки) или ВидРеквизита (ссылка на _ВидыРеквизитов). 
//         Значение - конкретное значение для сравнения, ссылка на _Карточки
//       либо:
//         Ключ - "Карточки" (ссылка на _Карточки) или ВидРеквизита (ссылка на _ВидыРеквизитов). 
//         Значение - массив карточек, ссылка на _Карточки
//       либо:
//         Ключ - ВидРеквизита (ссылка на _ВидыРеквизитов). 
//         Значение - конкретное значение для сравнения.
//     Условия по разным ВидРеквизитов объединяются по И.
//  "МенеджерВременныхТаблиц" - МенеджерВременныхТаблиц.
//      После выполнения функции в нем будут доступны таблицы ВТ_Реквизиты и ВТ_ЗначенияРеквизитов.
Функция ПолучитьСостояние(ВидКарточки = Неопределено, Отбор = Неопределено, МенеджерВременныхТаблиц = Неопределено) Экспорт

	// 1. Подготовка Менеджера Временных Таблиц
	Если Не ТипЗнч(МенеджерВременныхТаблиц) = Тип("МенеджерВременныхТаблиц") Тогда
		МенеджерВременныхТаблиц = Новый МенеджерВременныхТаблиц;
	КонецЕсли;

	// 2. Инициализация переменных для отборов
	МассивКарточек = Новый Массив;
	// Для отбора по конкретным значениям реквизитов
	ОтборыПоРеквизитам = Новый Соответствие; // Ключ - ВидРеквизита, Значение - ЗначениеРеквизита

	// 3. Обработка параметра "Отбор"
	Если Отбор <> Неопределено Тогда
		Для Каждого КлючЗнач Из Отбор Цикл
			Если КлючЗнач.Ключ = "Карточка"
			   И ТипЗнч(КлючЗнач.Значение) = Тип("СправочникСсылка._Карточки")
			Тогда
				// Проверяем, есть ли элемент в массиве. Функция Найти возвращает номер, если нашла, и Неопределено, если нет.
				НайденныйИндекс = МассивКарточек.Найти(КлючЗнач.Значение);
				Если НайденныйИндекс = Неопределено Тогда
					МассивКарточек.Добавить(КлючЗнач.Значение);
				КонецЕсли;
			ИначеЕсли КлючЗнач.Ключ = "Карточки"
				    И ТипЗнч(КлючЗнач.Значение) = Тип("Массив")
			Тогда
				Для Каждого Карточка из КлючЗнач.Значение Цикл 
					// Проверяем, есть ли элемент в массиве. Функция Найти возвращает номер, если нашла, и Неопределено, если нет.
					НайденныйИндекс = МассивКарточек.Найти(КлючЗнач.Значение);
					Если НайденныйИндекс = Неопределено Тогда
						МассивКарточек.Добавить(КлючЗнач.Значение);
					КонецЕсли;
				КонецЦикла;
			ИначеЕсли ТипЗнч(КлючЗнач.Ключ) = Тип("ПланВидовХарактеристикСсылка._ВидыРеквизитов") Тогда // Предполагаю, что ключ - это ВидРеквизита
				// Если ключ уже был, значит ошибка в отборе или последующий перезапишет предыдущий
				// Для простоты просто перезаписываем.
				ОтборыПоРеквизитам.Вставить(КлючЗнач.Ключ, КлючЗнач.Значение);
			КонецЕсли;
		КонецЦикла;
	КонецЕсли;

	// 4. Определение ВидКарточки
	Если ВидКарточки = Неопределено И МассивКарточек.Количество() > 0 Тогда
		// Берем первую карточку для определения Вида
		ПроизвольнаяКарточка = МассивКарточек[0];
		ВидКарточки = ПроизвольнаяКарточка.ВидКарточки;
	КонецЕсли;

	// Если ВидКарточки так и остался Неопределено, невозможно определить структуру.
	Если ВидКарточки = Неопределено Тогда
		ВызватьИсключение "Не задан и не может быть определен параметр ВидКарточки.";
	КонецЕсли;


	//5. Готовлю текст запроса
	//5.01 Формирую ВТ_Реквизиты (структура/шаблон вида карточки) в одном запросе
	ТекстЗапросаВТ_Реквизиты = "
		|////////////////////////////////////////////////////////////////////////////////
		|// Формируем ВТ_Реквизиты - шаблон реквизитов для ВидКарточки
		|////////////////////////////////////////////////////////////////////////////////
		|ВЫБРАТЬ
		|	Реквизиты.Ссылка КАК ВидКарточки,
		|	Реквизиты.ВидРеквизита КАК ВидРеквизита
		|ПОМЕСТИТЬ ВТ_Реквизиты
		|ИЗ
		|	Справочник._ВидыКарточек.Реквизиты КАК Реквизиты
		|ГДЕ
		|	Реквизиты.Ссылка = &ВидКарточки
		|;
		|";

	//5.10  
	// --- Подготовка к формированию ВТ_ЗначенияРеквизитов с учетом отбора ---
	// Если есть отборы по реквизитам, сначала нужно найти карточки, которые им удовлетворяют.
	// Для этого делаем внутренние запросы, чтобы получить список подходящих карточек.
	// Этот список будет использован в основном запросе для формирования ВТ_ЗначенияРеквизитов.
	
	ТекстЗапросаОтобранныхКарточекПоУсловиямОтбора = "";
	ТекстЗапросаНаУдалениеСлужебныхТаблиц ="";
	ПараметрыЗапроса = Новый Структура;
	ПараметрыЗапроса.Вставить("ВидКарточки", ВидКарточки);

	//5.11  выбираю доступные карточки, если таковые перечисленны в параметрах отбора
	Если МассивКарточек.Количество() > 0 Тогда
		УсловиеВТ_ЗначенияРеквизитов = "Карточка В (&МассивКарточек)";
		Условие_МассивКарточек = " И Карточка В (&МассивКарточек)";
		ПараметрыЗапроса.Вставить("МассивКарточек", МассивКарточек);
		
		ТекстЗапросаОтобранныхКарточекПоУсловиямОтбора = ТекстЗапросаОтобранныхКарточекПоУсловиямОтбора + "
			|ВЫБРАТЬ
			|	_Карточки.Ссылка КАК Карточка
			|ПОМЕСТИТЬ ВТ_Карточки
			|ИЗ
			|	Справочник._Карточки КАК _Карточки
			|ГДЕ
			|	_Карточки.Ссылка В (&МассивКарточек)
			|;";
		ВТ_ТаблицаОтобраныхКарточек = "ВТ_Карточки";
	Иначе
		УсловиеВТ_ЗначенияРеквизитов = "";
		Условие_МассивКарточек = "";
	КонецЕсли;
	
    //5.12 Отбираю карточки, сотласно указанным значениям реквизитов
 	Если ОтборыПоРеквизитам.Количество() > 0 Тогда
		ТекущаяВТ="";
		Сч=0;
		Для каждого ОтборПоРеквизиту ИЗ ОтборыПоРеквизитам Цикл
			Сч = Сч+1;
			Сч_ТекущаяВТ=Формат(Сч,"ЧГ=0"); //Использую Формат(), чтобы исключить ошибку перехода на "1 000"
			Сч_ПредыдущаяВТ=Формат(Сч-1,"ЧГ=0"); //Использую Формат(), чтобы исключить ошибку перехода на "1 000"
			
			ТекВидРеквизита = ОтборПоРеквизиту.Ключ; 
			ТекЗначениеРеквизита = ОтборПоРеквизиту.Значение;
			ПараметрыЗапроса.Вставить("ВидРеквизита"+Сч_ТекущаяВТ, ТекВидРеквизита);
			ПараметрыЗапроса.Вставить("ЗначениеРеквизита"+Сч_ТекущаяВТ,ТекЗначениеРеквизита);

			// Имя предыдущей таблицы
			ИмяПредыдущейТаблицы = "ВТ_КарточкиОтбора_" + Сч_ПредыдущаяВТ;
			// Имя новой таблицы
			ИмяТекущейТаблицы = "ВТ_КарточкиОтбора_" + Сч_ТекущаяВТ;

			ТекстЗапросаОтобранныхКарточекПоУсловиямОтбора = ТекстЗапросаОтобранныхКарточекПоУсловиямОтбора + "
			|////////////////////////////////////////////////////////////////////////////////
			|// Формируем ВТ_КарточкиОтбора_" + Сч_ТекущаяВТ + " - пересечение с " + Строка(ТекВидРеквизита) + "
			|////////////////////////////////////////////////////////////////////////////////
			|ВЫБРАТЬ
			|	Регистр.Карточка КАК Карточка
			|ПОМЕСТИТЬ " + ИмяТекущейТаблицы + "
			|ИЗ
			|	РегистрСведений._РеквизитыКарточек.СрезПоследних(,
			|		ВидКарточки = &ВидКарточки И ВидРеквизита = &ВидРеквизита"+Сч_ТекущаяВТ + "
			|		"+Условие_МассивКарточек+"
			|	) КАК Регистр";
			Если Сч>1 Тогда  // Отбор по подготовленной 
				ТекстЗапросаОтобранныхКарточекПоУсловиямОтбора = ТекстЗапросаОтобранныхКарточекПоУсловиямОтбора + "
				|	ВНУТРЕННЕЕ СОЕДИНЕНИЕ " + ИмяПредыдущейТаблицы + " КАК ПредыдущаяТаблица
				|	ПО (ПредыдущаяТаблица.Карточка = Регистр.Карточка)";
			КонецЕсли;
			ТекстЗапросаОтобранныхКарточекПоУсловиямОтбора = ТекстЗапросаОтобранныхКарточекПоУсловиямОтбора + "
			|ГДЕ
			|	Регистр.ЗначениеРеквизита = &ЗначениеРеквизита" + Сч_ТекущаяВТ + "
			|;";
			
			ТекстЗапросаНаУдалениеСлужебныхТаблиц = ТекстЗапросаНаУдалениеСлужебныхТаблиц + "
			|УНИЧТОЖИТЬ ВТ_КарточкиОтбора_" + Сч_ТекущаяВТ + ";";

		КонецЦикла;

		// Теперь у нас есть финальная таблица с отобранными карточками
		// Имя финальной таблицы
		ВТ_ТаблицаОтобраныхКарточек = ИмяТекущейТаблицы;
	КонецЕсли; // Конец обработки отборов по реквизитам

	//5.13 Результат отбора карточек всегда будет лежать в ВТ_КарточкиОтбораИтог
	// так же умножаю карточки на получаемые реквизиты
	ТекстЗапросаОтобранныхКарточекПоУсловиямОтбора = ТекстЗапросаОтобранныхКарточекПоУсловиямОтбора + "
			|////////////////////////////////////////////////////////////////////////////////
			|// Формируем ВТ_КарточкиОтбораИтог - итоговый список карточек после всех отборов
			|////////////////////////////////////////////////////////////////////////////////
			|ВЫБРАТЬ

			|	" + ВТ_ТаблицаОтобраныхКарточек + ".Карточка,
			|	ВТ_Реквизиты.ВидКарточки КАК ВидКарточки,
			|	ВТ_Реквизиты.ВидРеквизита КАК ВидРеквизита
			|ПОМЕСТИТЬ ВТ_КарточкиОтбораИтог
			|ИЗ
			|	" + ВТ_ТаблицаОтобраныхКарточек + "
			|	ЛЕВОЕ СОЕДИНЕНИЕ ВТ_Реквизиты
			|	ПО (ИСТИНА)
			|;";
	УсловиеВТ_ЗначенияРеквизитов = "Карточка В (ВЫБРАТЬ Карточка ИЗ ВТ_КарточкиОтбораИтог)";

	ТекстЗапросаНаУдалениеСлужебныхТаблиц = ТекстЗапросаНаУдалениеСлужебныхТаблиц + "
			|УНИЧТОЖИТЬ ВТ_КарточкиОтбораИтог;";

	
	//5.20 Формирование ВТ_Реквизиты (структура/шаблон вида карточки) и ВТ_ЗначенияРеквизитов в одном запросе
	ТекстЗапросаВТ_ЗначенияРеквизитов = "
		|////////////////////////////////////////////////////////////////////////////////
		|// Формируем ВТ_ЗначенияРеквизитов - значения для найденных карточек
		|// ВТ_ЗначенияРеквизитов строится на основе ВТ_Реквизиты, т.е. содержит все возможные
		|// реквизиты для ВидКарточки, но только для карточек, прошедших отбор.
		|////////////////////////////////////////////////////////////////////////////////
		|ВЫБРАТЬ
		|	ВТ_КарточкиОтбораИтог.Карточка КАК Карточка,
		|	ВТ_КарточкиОтбораИтог.ВидРеквизита КАК ВидРеквизита,
		|	Регистр.УИД_Элемента КАК УИД_Элемента,
		|	Регистр.ЗначениеРеквизита КАК ЗначениеРеквизита
		|ПОМЕСТИТЬ ВТ_ЗначенияРеквизитовКарточек
		|ИЗ
		|	ВТ_КарточкиОтбораИтог КАК ВТ_КарточкиОтбораИтог
		|		ЛЕВОЕ СОЕДИНЕНИЕ РегистрСведений._РеквизитыКарточек.СрезПоследних(,"+ УсловиеВТ_ЗначенияРеквизитов +") КАК Регистр
		|		ПО ВТ_КарточкиОтбораИтог.Карточка = Регистр.Карточка
		|		 И ВТ_КарточкиОтбораИтог.ВидРеквизита = Регистр.ВидРеквизита
		|;
		|";


	// 9. Формирование возвращаемого массива структур
	//    Выбираем данные из уже созданной временной таблицы ВТ_ЗначенияРеквизитов
	Запрос = Новый Запрос;
	Запрос.МенеджерВременныхТаблиц = МенеджерВременныхТаблиц;
	Для Каждого Пар Из ПараметрыЗапроса Цикл
		Запрос.УстановитьПараметр(Пар.Ключ, Пар.Значение);
	КонецЦикла;
	Запрос.Текст = " 
		|" + ТекстЗапросаВТ_Реквизиты + "
		|" + ТекстЗапросаОтобранныхКарточекПоУсловиямОтбора + "
		|" + ТекстЗапросаВТ_ЗначенияРеквизитов + "
		|" + ТекстЗапросаНаУдалениеСлужебныхТаблиц + " 
		|
		|////////////////////////////////////////////////////////////////////////////////
		|ВЫБРАТЬ
		|	*
		|ИЗ
		|	ВТ_ЗначенияРеквизитовКарточек
		|УПОРЯДОЧИТЬ ПО Карточка,ВидРеквизита 
		|";

	Выборка = Запрос.Выполнить().Выбрать();
	
	Результат = Новый Массив;
	Пока Выборка.Следующий() Цикл
		ЭлементРезультата = Новый Структура("Карточка,ВидРеквизита,УИД_Элемента,ЗначениеРеквизита");
		ЗаполнитьЗначенияСвойств(ЭлементРезультата,Выборка);
		Результат.Добавить(ЭлементРезультата);
	КонецЦикла;

	// Функция завершена, возвращаем результат.
	// Временные таблицы "ВТ_Реквизиты" и "ВТ_ЗначенияРеквизитов" остаются в МенеджереВременныхТаблиц
	// и могут быть использованы вызывающим кодом.
	Возврат Результат;

КонецФункции
